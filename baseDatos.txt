CREATE DATABASE economark;
USE economark;

-- Tabla de credenciales
CREATE TABLE credenciales (
	id INT AUTO_INCREMENT NOT NULL PRIMARY KEY,
	rol varchar(80) NOT NULL, 
	username varchar(80) NOT NULL,
	password varchar(300) NOT NULL,
  	createdAt DATE,
  	updatedAt DATE
);

-- Tabla de categorías
CREATE TABLE categoria (
	id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
	idCategoria varchar(50) NOT NULL,
	nombreCategoria varchar(50) NOT NULL UNIQUE,
	descripcionCategoria varchar(256) NULL,
	estado BIT DEFAULT 1,
	createdAt DATE,
  	updatedAt DATE

-- Tabla de productos (se mantiene)
CREATE TABLE productos (
	id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
	nombre varchar(80) NOT NULL,
	categoria varchar(150) NOT NULL,
	precio DECIMAL(10,3) NOT NULL,
	createdAt DATE,
	updatedAt DATE

-- Tabla de artículos
CREATE TABLE articulo (
	idArticulo INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
	idCategoria INT NOT NULL,
	codigo varchar(50) NULL,
	nombre varchar(100) NOT NULL UNIQUE,
	precio_venta DECIMAL(11,2) NOT NULL,
	stock INT NOT NULL,
	descripcion varchar(256) NULL,
	estado BIT DEFAULT 1,
	FOREIGN KEY (idCategoria) REFERENCES categoria(id)
);

-- Tabla de personas (clientes y proveedores)
CREATE TABLE persona (
	idPersona INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
	tipo_persona varchar(20) NOT NULL,
	nombre varchar(100) NOT NULL,
	tipo_documento varchar(20) NULL,
	num_documento varchar(20) NULL,
	direccion varchar(70) NULL,
	telefono varchar(20) NULL,
	email varchar(50) NULL
);

-- Tabla de ingresos
CREATE TABLE ingreso (
	idIngreso INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
	idProveedor INT NOT NULL,
	idUsuario INT NOT NULL,
	tipo_comprobante varchar(20) NOT NULL,
	serie_comprobante varchar(7) NULL,
	num_comprobante varchar(10) NOT NULL,
	fecha DATETIME NOT NULL,
	impuesto DECIMAL(4,2) NOT NULL,
	total DECIMAL(11,2) NOT NULL,
	estado varchar(20) NOT NULL,
	FOREIGN KEY (idProveedor) REFERENCES persona(idPersona),
	FOREIGN KEY (idUsuario) REFERENCES credenciales(id)
);

-- Detalle de ingreso
CREATE TABLE detalle_ingreso (
	idDetalle_ingreso INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
	idIngreso INT NOT NULL,
	idArticulo INT NOT NULL,
	cantidad INT NOT NULL,
	precio DECIMAL(11,2) NOT NULL,
	FOREIGN KEY (idIngreso) REFERENCES ingreso(idIngreso) ON DELETE CASCADE,
	FOREIGN KEY (idArticulo) REFERENCES articulo(idArticulo)
);

-- Tabla de ventas
CREATE TABLE venta (
	idVenta INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
	idCliente INT NOT NULL,
	idUsuario INT NOT NULL,
	tipo_comprobante varchar(20) NOT NULL,
	serie_comprobante varchar(7) NULL,
	num_comprobante varchar(10) NOT NULL,
	fecha_hora DATETIME NOT NULL,
	impuesto DECIMAL(4,2) NOT NULL,
	total DECIMAL(11,2) NOT NULL,
	estado varchar(20) NOT NULL,
	FOREIGN KEY (idCliente) REFERENCES persona(idPersona),
	FOREIGN KEY (idUsuario) REFERENCES credenciales(id)
);

-- Detalle de venta
CREATE TABLE detalle_venta (
	idDetalle_venta INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
	idVenta INT NOT NULL,
	idArticulo INT NOT NULL,
	cantidad INT NOT NULL,
	precio DECIMAL(11,2) NOT NULL,
	descuento DECIMAL(11,2) NOT NULL,
	FOREIGN KEY (idVenta) REFERENCES venta(idVenta) ON DELETE CASCADE,
	FOREIGN KEY (idArticulo) REFERENCES articulo(idArticulo)
);

-- Proveedores
CREATE TABLE proveedores (
    idProveedor INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    telefono VARCHAR(20) NOT NULL,
    email VARCHAR(100) NOT NULL,
    direccion VARCHAR(150) NOT NULL,
    createdAt DATETIME DEFAULT CURRENT_TIMESTAMP,
    updatedAt DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);
